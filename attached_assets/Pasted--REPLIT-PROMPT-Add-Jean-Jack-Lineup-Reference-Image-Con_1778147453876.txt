### 2. Run this script to combine into one lineup image

```python
from PIL import Image
import os
import glob

temp_dir = 'backend/photo_report/references/temp'
refs_dir = 'backend/photo_report/references'

# Find all JJ images in temp (any extension)
files = sorted(glob.glob(os.path.join(temp_dir, 'JJ_*')))
if not files:
    files = sorted(glob.glob(os.path.join(temp_dir, 'jj_*')))
    
print(f"Found {len(files)} Jean Jack images:")
for f in files:
    print(f"  {os.path.basename(f)}")

# Load all images
imgs = [Image.open(f) for f in files]

# Resize all to same height (1200px)
target_h = 1200
resized = []
for img in imgs:
    ratio = target_h / img.height
    new_w = int(img.width * ratio)
    resized.append(img.resize((new_w, target_h), Image.LANCZOS))

# Add spacing between bottles
spacing = 20
total_w = sum(img.width for img in resized) + spacing * (len(resized) - 1)

# Create lineup on white background
lineup = Image.new('RGB', (total_w, target_h), (255, 255, 255))
x = 0
for i, img in enumerate(resized):
    lineup.paste(img, (x, 0))
    x += img.width + spacing

# Save as optimized JPEG
out_path = os.path.join(refs_dir, 'jeanjack_lineup.jpg')

# Resize if too wide (max 2400px to save tokens)
if lineup.width > 2400:
    ratio = 2400 / lineup.width
    lineup = lineup.resize((2400, int(lineup.height * ratio)), Image.LANCZOS)

lineup.save(out_path, format="JPEG", quality=85)
print(f"\nSaved lineup: {out_path} ({lineup.size[0]}x{lineup.size[1]})")

# Clean up temp files
for f in files:
    os.remove(f)
    print(f"Removed temp: {os.path.basename(f)}")

# Remove temp dir if empty
if not os.listdir(temp_dir):
    os.rmdir(temp_dir)
    print("Removed temp directory")
```
