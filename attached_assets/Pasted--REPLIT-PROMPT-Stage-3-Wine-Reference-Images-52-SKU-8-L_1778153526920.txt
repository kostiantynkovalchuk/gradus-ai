def find_file(name):
    """Find file by name in temp dir, case-insensitive, any extension."""
    # Try exact match first
    for ext in ['.png', '.jpg', '.jpeg', '.PNG', '.JPG']:
        path = os.path.join(TEMP_DIR, name + ext)
        if os.path.exists(path):
            return path
    # Fallback: case-insensitive scan
    name_lower = name.lower()
    for f in os.listdir(TEMP_DIR):
        base = os.path.splitext(f)[0]
        if base.lower() == name_lower:
            return os.path.join(TEMP_DIR, f)
    return None


def create_lineup(file_names, output_name):
    """Load images by name, combine into single lineup image."""
    print(f"\n=== Creating {output_name} ({len(file_names)} SKU) ===")
    
    imgs = []
    for name in file_names:
        path = find_file(name)
        if path:
            try:
                img = Image.open(path)
                imgs.append(img)
                print(f"  OK {name}")
            except Exception as e:
                print(f"  ERROR {name}: {e}")
        else:
            print(f"  NOT FOUND: {name}")
    
    if not imgs:
        print(f"  SKIPPED — no images found")
        return
    
    # Resize all to same height
    resized = []
    for img in imgs:
        ratio = TARGET_H / img.height
        resized.append(img.resize((int(img.width * ratio), TARGET_H), Image.LANCZOS))
    
    # Combine horizontally
    total_w = sum(i.width for i in resized) + SPACING * (len(resized) - 1)
    lineup = Image.new('RGB', (total_w, TARGET_H), (255, 255, 255))
    x = 0
    for img in resized:
        lineup.paste(img, (x, 0))
        x += img.width + SPACING
    
    # Resize if too wide
    if lineup.width > MAX_W:
        ratio = MAX_W / lineup.width
        lineup = lineup.resize((MAX_W, int(lineup.height * ratio)), Image.LANCZOS)
    
    out_path = os.path.join(REFS_DIR, output_name)
    lineup.save(out_path, format="JPEG", quality=85)
    size_kb = os.path.getsize(out_path) // 1024
    print(f"  SAVED: {output_name} ({lineup.size[0]}x{lineup.size[1]}, {size_kb}KB)")


# === CHECK FILES ===
print("=== Checking temp directory ===")
if not os.path.exists(TEMP_DIR):
    print(f"ERROR: {TEMP_DIR} does not exist!")
    exit(1)

temp_files = os.listdir(TEMP_DIR)
print(f"Found {len(temp_files)} files in temp/")

# === CREATE ALL LINEUPS ===
for output_name, file_names in LINEUP_GROUPS.items():
    create_lineup(file_names, output_name)

# === CLEANUP ===
print("\n=== Cleaning up ===")
for f in os.listdir(TEMP_DIR):

for f in all_refs:
