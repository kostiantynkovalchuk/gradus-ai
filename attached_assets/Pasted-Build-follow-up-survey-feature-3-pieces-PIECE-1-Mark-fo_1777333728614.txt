The job function:

async def _pulse_followup_check():
    """Send follow-up surveys to employees 14+ days after HR resolved their issue"""
    db = _models.SessionLocal()
    try:
        # Find employees due for follow-up
        rows = db.execute(sa_text("""
            SELECT ps.id, ps.telegram_id, ps.employee_name
            FROM pulse_surveys ps
            JOIN pulse_hr_actions pha ON pha.trigger_id IN (
                SELECT pt.id FROM pulse_triggers pt 
                WHERE pt.employee_id = ps.telegram_id
                AND pt.fired_at >= ps.responded_at - INTERVAL '1 day'
            )
            WHERE ps.score = 1
              AND ps.followup_scheduled = TRUE
              AND ps.followup_sent_at IS NULL
              AND pha.action = 'resolved'
              AND pha.created_at < NOW() - INTERVAL '14 days'
            LIMIT 10
        """)).fetchall()
        
        for row in rows:
            survey_id, tid, name = row
            await send_followup_survey(tid)
            db.execute(sa_text(
                "UPDATE pulse_surveys SET followup_sent_at = NOW() WHERE id = :sid"
            ), {"sid": survey_id})
            db.commit()
            logger.info(f"[PULSE] Follow-up survey sent to {name} ({tid})")
    except Exception as e:
        logger.error(f"[PULSE] Follow-up check failed: {e}")
        db.rollback()
    finally:
        db.close()
