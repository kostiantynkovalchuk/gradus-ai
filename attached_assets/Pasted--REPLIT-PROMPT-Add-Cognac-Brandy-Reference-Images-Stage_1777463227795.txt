### 3. Resize uploaded images

After I upload the 7 files, resize any that are over 1200px:
- Max 1200px on longest side
- Convert PNG → JPEG quality 85
- Save as `.jpg` in references/

```python
from PIL import Image
from io import BytesIO
import os
import glob

refs_dir = 'backend/photo_report/references'

# Process all new cognac images
for pattern in ['adjari_*.png', 'adjari_*.jpg', 'dovbush_*.png', 'dovbush_*.jpg']:
    for src in glob.glob(os.path.join(refs_dir, pattern)):
        img = Image.open(src)
        name = os.path.splitext(os.path.basename(src))[0]
        dst = os.path.join(refs_dir, f'{name}.jpg')
        
        if max(img.size) > 1200:
            ratio = 1200 / max(img.size)
            new_size = (int(img.size[0] * ratio), int(img.size[1] * ratio))
            img = img.resize(new_size, Image.LANCZOS)
        
        img.convert("RGB").save(dst, format="JPEG", quality=85)
        print(f"Saved {dst}: {img.size}")
        
        # Remove source PNG if we created a JPG
        if src != dst and os.path.exists(src):
            os.remove(src)
            print(f"Removed {src}")
```
