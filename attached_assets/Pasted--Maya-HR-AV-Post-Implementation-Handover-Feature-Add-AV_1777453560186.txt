# Maya HR — Архів AV Post Implementation Handover

**Feature:** Add "📰 Архів AV Post" top-level menu button to Maya HR with auto-archiving of new editions detected from broadcasts.

**Detection rule:** Any broadcast (text or photo caption) containing a URL matching `avpost-bestbrands.vercel.app/*` is automatically archived as a new edition.

**First edition (seed):** `https://avpost-bestbrands.vercel.app/` — title "Випуск №1 — Best Brands"

---

## 1. Architecture Decisions

| Decision | Choice | Reason |
|---|---|---|
| Detection | URL regex on `avpost-bestbrands.vercel.app/*` | Zero-friction — Natalia broadcasts as usual, archive auto-fills |
| Menu placement | Top-level button (alongside HR categories) | Highest visibility for company newspaper |
| Storage | New table `hr_avpost_editions` | Decoupled from `hr_broadcast_log` — survives broadcast log purges, allows manual edits |
| Hook point | `execute_broadcast()` in `broadcast_service.py` — AFTER successful send | Only archive editions that actually went out; failed broadcasts are not archived |
| Display | Inline keyboard list, newest first, paginated 5 per page | Consistent with existing Maya menu patterns |
| Idempotency | `UNIQUE` constraint on `url` | Re-broadcasting same edition won't create duplicates |

## 3. New Service — `avpost_service.py`

**File:** `backend/services/avpost_service.py`

```python
"""
AV Post archive service.

Responsibilities:
1. Detect AV Post URLs in broadcast content (text + captions)
2. Auto-archive new editions after successful broadcast
3. Provide list/get APIs for the Maya menu
"""
import re
import logging
import psycopg2
from typing import Optional
from datetime import datetime
from backend.db import get_db_url  # adjust import to match your project's db helper

logger = logging.getLogger(__name__)

# Matches https://avpost-bestbrands.vercel.app and any path/query
AVPOST_URL_REGEX = re.compile(
    r'https?://(?:www\.)?avpost-bestbrands\.vercel\.app(?:/[^\s\)\]]*)?',
    re.IGNORECASE
)


def extract_avpost_url(text: Optional[str]) -> Optional[str]:
    """Return the first AV Post URL found in text, or None."""
    if not text:
        return None
    match = AVPOST_URL_REGEX.search(text)
    if not match:
        return None
    url = match.group(0).rstrip('.,;:!?)')  # strip trailing punctuation
    return url


def _next_edition_number(cur) -> int:
    cur.execute("SELECT COALESCE(MAX(edition_number), 0) + 1 FROM hr_avpost_editions")
    return cur.fetchone()[0]


def archive_edition(
    url: str,
    broadcast_id: Optional[int] = None,
    source: str = 'broadcast',
    title: Optional[str] = None,
) -> Optional[int]:
    """
    Insert a new edition. Returns the edition id, or None if URL already archived.
    Idempotent — duplicate URLs return the existing id.
    """
    conn = psycopg2.connect(get_db_url())
    try:
        with conn:
            with conn.cursor() as cur:
                # Already archived?
                cur.execute(
                    "SELECT id FROM hr_avpost_editions WHERE url = %s",
                    (url,)
                )
                row = cur.fetchone()
                if row:
                    logger.info(f"AV Post edition already archived: {url} (id={row[0]})")
                    return row[0]

                edition_num = _next_edition_number(cur)
                final_title = title or f"Випуск №{edition_num}"

                cur.execute("""
                    INSERT INTO hr_avpost_editions
                        (url, title, edition_number, source, broadcast_id, published_at)
                    VALUES (%s, %s, %s, %s, %s, NOW())
                    RETURNING id
                """, (url, final_title, edition_num, source, broadcast_id))
                edition_id = cur.fetchone()[0]
                logger.info(
                    f"📰 AV Post edition archived: #{edition_num} {url} (id={edition_id})"
                )
                return edition_id
    finally:
        conn.close()


def list_editions(limit: int = 50, offset: int = 0) -> list[dict]:
    """Return active editions, newest first."""
    conn = psycopg2.connect(get_db_url())
    try:
        with conn.cursor() as cur:
            cur.execute("""
                SELECT id, url, title, edition_number, published_at
                FROM hr_avpost_editions
                WHERE is_active = TRUE
                ORDER BY published_at DESC, id DESC
                LIMIT %s OFFSET %s
            """, (limit, offset))
            return [
                {
                    'id': r[0],
                    'url': r[1],
                    'title': r[2],
                    'edition_number': r[3],
                    'published_at': r[4],
                }
                for r in cur.fetchall()
            ]
    finally:
        conn.close()


def count_editions() -> int:
    conn = psycopg2.connect(get_db_url())
    try:
        with conn.cursor() as cur:
            cur.execute("SELECT COUNT(*) FROM hr_avpost_editions WHERE is_active = TRUE")
            return cur.fetchone()[0]
    finally:
        conn.close()


def maybe_archive_from_broadcast(
    text_content: Optional[str],
    caption: Optional[str],
    broadcast_id: Optional[int],
) -> Optional[int]:
    """
    Called from broadcast_service.execute_broadcast AFTER successful send.
    Scans text + caption for an AV Post URL. If found, archives it.
    Safe to call on every broadcast — non-AV-Post broadcasts are no-ops.
    """
    url = extract_avpost_url(text_content) or extract_avpost_url(caption)
    if not url:
        return None
    try:
        return archive_edition(url=url, broadcast_id=broadcast_id, source='broadcast')
    except Exception as e:
        # Never let archive failures break the broadcast flow
        logger.exception(f"Failed to archive AV Post edition from broadcast {broadcast_id}: {e}")
        return None
```

**Why psycopg2, not SQLAlchemy:** Matches the pattern already established in Maya Hunt for background-task DB ops (see `Maya_Hunt_Architecture.docx` §3.5 — SQLAlchemy returned empty results from background tasks). The broadcast execution runs as a background task too.

Then implement the handler in the same file (or a dedicated `backend/handlers/avpost_handler.py`):

```python
from backend.services import avpost_service

PAGE_SIZE = 5

async def handle_avpost_callback(query, data: str):
    """
    Routes:
        avpost:list:<offset>   — show editions list page
        avpost:back            — return to main menu
    """
    parts = data.split(":")
    action = parts[1] if len(parts) > 1 else "list"

    if action == "list":
        offset = int(parts[2]) if len(parts) > 2 else 0
        await render_editions_list(query, offset)
    elif action == "back":
        await render_main_menu(query)  # reuse existing main menu renderer


async def render_editions_list(query, offset: int):
    total = avpost_service.count_editions()
    editions = avpost_service.list_editions(limit=PAGE_SIZE, offset=offset)

    if not editions:
        text = "📰 *Архів AV Post*\n\nПоки що немає випусків."
        keyboard = [[InlineKeyboardButton("⬅ Назад", callback_data="avpost:back")]]
        await query.edit_message_text(
            text,
            parse_mode="Markdown",
            reply_markup=InlineKeyboardMarkup(keyboard),
        )
        return

    lines = [
        "📰 *Архів AV Post*",
        f"_Усього випусків: {total}_",
        "",
    ]
    keyboard = []
    for ed in editions:
        date_str = ed['published_at'].strftime('%d.%m.%Y')
        label = f"№{ed['edition_number']} · {date_str} — {ed['title']}"
        # URL buttons open externally — no callback needed
        keyboard.append([InlineKeyboardButton(label, url=ed['url'])])

    # Pagination row
    nav = []
    if offset > 0:
        prev_offset = max(0, offset - PAGE_SIZE)
        nav.append(InlineKeyboardButton("⬅", callback_data=f"avpost:list:{prev_offset}"))
    if offset + PAGE_SIZE < total:
        next_offset = offset + PAGE_SIZE
        nav.append(InlineKeyboardButton("➡", callback_data=f"avpost:list:{next_offset}"))
    if nav:
        keyboard.append(nav)

    keyboard.append([InlineKeyboardButton("🏠 Головне меню", callback_data="avpost:back")])

    await query.edit_message_text(
        "\n".join(lines),
        parse_mode="Markdown",
        reply_markup=InlineKeyboardMarkup(keyboard),
        disable_web_page_preview=True,
    )
```

**Note on `url=` buttons:** Telegram inline-keyboard buttons with a `url` field open the link directly when tapped — no callback round-trip, opens in user's browser or Telegram's in-app browser. This is the cleanest UX for a magazine archive.
