```python
def ingest_law(law_code: str) -> dict:
    """
    Fetch a law from zakon.rada.gov.ua, parse current edition, chunk by article,
    embed, upsert to Pinecone with full metadata.

    Returns: {chunks_added, edition_date, edition_basis, next_edition_basis}
    """
    source = db.session.query(SolomonKbSource).filter_by(law_code=law_code).first()
    if not source:
        raise ValueError(f"Law {law_code} not in approved registry")
    if source.status != 'active':
        raise ValueError(f"Law {law_code} has status {source.status}, not active")

    # 1. Fetch canonical URL, follow "не поточна редакція" redirect if present
    html, final_url = _fetch_current_edition(source.canonical_url, max_hops=3)

    # 2. Parse header
    edition_date, edition_basis, next_basis, next_date = _parse_edition_header(html)

    # 3. Chunk by article (Стаття N. ...)
    chunks = _chunk_by_article(html, source_id=source.id, source_title=source.law_name, official_url=source.canonical_url + "/print1")

    # 4. Embed via OpenAI
    embeddings = embed_chunks([c["chunk_text"] for c in chunks])

    # 5. Delete any existing chunks for this source (for re-ingest scenarios)
    _delete_existing_chunks(source.canonical_url)

    # 6. Upsert with metadata
    vectors = [
        {
            "id": f"corpus_{source.id}_{i}",
            "values": emb,
            "metadata": {
                **chunks[i],
                "kb_source_id": source.id,
                "current_edition_date": edition_date.isoformat() if edition_date else None,
            }
        }
        for i, emb in enumerate(embeddings)
    ]
    index.upsert(vectors=vectors, namespace=NAMESPACE)

    # 7. Update registry
    source.current_edition_date = edition_date
    source.current_edition_basis = edition_basis
    source.next_edition_basis = next_basis
    source.next_edition_date_estimated = next_date
    source.last_verified_at = datetime.now()
    db.session.commit()

    return {"chunks_added": len(vectors), "edition_date": edition_date, "edition_basis": edition_basis}


def _fetch_current_edition(canonical_url: str, max_hops: int) -> tuple[str, str]:
    """Fetch URL, detect 'не поточна редакція' warning, follow to current. Returns (html, final_url)."""
    url = canonical_url
    for _ in range(max_hops):
        resp = requests.get(url, timeout=30)
        resp.raise_for_status()
        soup = BeautifulSoup(resp.text, "html.parser")
        warning = soup.find(string=re.compile(r"Це не поточна редакція документу"))
        if not warning:
            return resp.text, url
        # Find "Перейти до поточної" link
        link = soup.find("a", string=re.compile(r"Перейти до поточної"))
        if not link or not link.get("href"):
            return resp.text, url  # accept what we have
        url = _absolutize(link["href"])
    return resp.text, url
```

### Filter function

`backend/solomon/contracts/citation_filter.py`:

```python
def normalize_url(url: str) -> str:
    """Strip /print1 suffix, /edYYYYMMDD suffix, and #fragment."""
    url = re.sub(r"/print1.*$", "", url)
    url = re.sub(r"/ed\d{8}.*$", "", url)
    url = url.split("#")[0]
    return url.rstrip("/")


def filter_citations(citations: list[dict]) -> tuple[list[dict], list[dict]]:
    """
    Returns (kept, dropped). Each dropped item: {article_ref, official_url, reason}.
    """
    approved_urls = {
        normalize_url(s.canonical_url)
        for s in get_active_kb_sources()
    }
    kept, dropped = [], []
    for cit in citations:
        if normalize_url(cit.get("official_url", "")) in approved_urls:
            kept.append(cit)
        else:
            dropped.append({
                **cit,
                "reason": f"Source not in approved KB registry"
            })
    return kept, dropped
```

```python
    finding.legal_citations = kept
    if not kept and raw_citations:
        finding.grounding_status = 'out_of_approved_kb'
        # The analyzer originally produced citations but none were approved — surface this honestly.
        # `proposed_alternative` text stays as-is; the UI's existing "Альтернатива не обґрунтована" fallback renders.
except Exception as e:
    logger.exception("Citation filter failed")
    finding.citation_filter_failed = True
    finding.legal_citations = raw_citations  # preserve original to not lose data
```

```python
def run_daily_edition_check():
    sources = db.session.query(SolomonKbSource).filter_by(status='active').all()
    for source in sources:
        try:
            check_one_source(source)
        except Exception as e:
            logger.exception(f"Edition check failed for {source.law_code}")
            _notify_dev_channel(f"Edition check failed for {source.law_code}: {e}")


def check_one_source(source: SolomonKbSource):
    history = SolomonKbSourceHistory(
        kb_source_id=source.id,
        edition_date_from=source.current_edition_date,
        edition_basis_from=source.current_edition_basis,
        status='in_progress',
        reingest_started_at=datetime.now(),
    )
    db.session.add(history)
    db.session.commit()

    try:
        # Fetch and parse header WITHOUT triggering full reingest
        html, _ = _fetch_current_edition(source.canonical_url, max_hops=3)
        new_date, new_basis, next_basis, _ = _parse_edition_header(html)

        if new_date == source.current_edition_date and new_basis == source.current_edition_basis:
            history.status = 'no_change'
            source.last_verified_at = datetime.now()
            db.session.commit()
            return

        # Edition flipped — full reingest
        result = ingest_law(source.law_code)  # this also updates source row
        history.edition_date_to = result['edition_date']
        history.edition_basis_to = result['edition_basis']
        history.chunks_added = result['chunks_added']
        history.status = 'completed'
        history.reingest_completed_at = datetime.now()
        db.session.commit()

    except Exception as e:
        history.status = 'failed'
        history.error_message = str(e)
        history.reingest_completed_at = datetime.now()
        db.session.commit()
        raise
```
