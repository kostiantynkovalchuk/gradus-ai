pythondef _format_reply_with_corrections(reply: str, errors: list) -> str:
    """Bold the corrected words in Sara's reply.
    
    Strategy: for each error, find words present in `correction` 
    but absent from `original` (the actual fix). Bold those 
    individual words wherever they appear in the reply.
    """
    if not reply:
        return ""
    if not errors:
        return html_escape(reply)
    try:
        # Collect the changed words across all errors
        bold_words = set()
        for err in errors:
            orig = (err.get("original") or "").lower().split()
            corr = (err.get("correction") or "").lower().split()
            orig_set = set(orig)
            # Words in correction that aren't in original = the actual fixes
            for w in corr:
                if w not in orig_set and len(w) > 1:
                    bold_words.add(w)
        if not bold_words:
            return html_escape(reply)
        # Build the result: bold matching words (case-insensitive)
        escaped = html_escape(reply)
        for word in sorted(bold_words, key=len, reverse=True):
            import re as _re
            pattern = _re.compile(
                r'(?<![&\w])(' + _re.escape(html_escape(word)) + r')(?!\w)',
                _re.IGNORECASE
            )
            escaped = pattern.sub(r'<b>\1</b>', escaped)
        return escaped
    except Exception:
        return html_escape(reply)
Key behaviors:

Compares original vs correction word sets — words in correction but not in original are the fixes (e.g. original "buy" → correction "bought" → bold word is "bought").
Skips single-character words (articles, etc.) to avoid noisy bolding.
Case-insensitive match against Sara's reply, so "Bought" at start of sentence still bolds.
Word-boundary aware (won't bold "ought" inside "bought").
Falls back to plain escaped text on any error.
Move the import re to the top of the file if not already there (it likely is — check).
