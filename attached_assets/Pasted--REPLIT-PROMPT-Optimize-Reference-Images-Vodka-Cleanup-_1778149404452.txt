### Step 1: Create Cognac Lineups from existing files

Run this Python script in the backend directory:

```python
from PIL import Image
import os

refs = 'backend/photo_report/references'

def create_lineup(file_list, output_name, target_h=1200, max_w=2400):
    """Combine multiple product photos into one lineup image."""
    imgs = []
    for f in file_list:
        path = os.path.join(refs, f)
        if os.path.exists(path):
            imgs.append(Image.open(path))
            print(f"  Loaded {f}")
        else:
            print(f"  WARNING: {f} not found, skipping")
    
    if not imgs:
        print(f"  ERROR: No images found for {output_name}")
        return
    
    # Resize all to same height
    resized = []
    for img in imgs:
        ratio = target_h / img.height
        resized.append(img.resize((int(img.width * ratio), target_h), Image.LANCZOS))
    
    # Combine with spacing
    spacing = 20
    total_w = sum(i.width for i in resized) + spacing * (len(resized) - 1)
    lineup = Image.new('RGB', (total_w, target_h), (255, 255, 255))
    x = 0
    for img in resized:
        lineup.paste(img, (x, 0))
        x += img.width + spacing
    
    # Resize if too wide
    if lineup.width > max_w:
        ratio = max_w / lineup.width
        lineup = lineup.resize((max_w, int(lineup.height * ratio)), Image.LANCZOS)
    
    out_path = os.path.join(refs, output_name)
    lineup.save(out_path, format="JPEG", quality=85)
    print(f"  Created {output_name}: {lineup.size[0]}x{lineup.size[1]}")


print("=== Creating ADJARI lineup ===")
create_lineup(
    ['adjari_3star_500.jpg', 'adjari_5star_tubus.jpg', 'adjari_7star_500.jpg'],
    'adjari_lineup.jpg'
)

print("\n=== Creating DOVBUSH lineup ===")
create_lineup(
    ['dovbush_3star_500.jpg', 'dovbush_4star_500.jpg', 'dovbush_4star_round.jpg', 'dovbush_honey_500.jpg'],
    'dovbush_lineup.jpg'
)

print("\n=== Deleting individual files ===")
to_delete = [
    # Vodka flagships (lineup already covers them)
    'greenday_classic_500.jpg',
    'helsinki_ice_palace_500.jpg',
    'ukrainka_traditional_500.jpg',
    # Cognac individuals (replaced by lineups)
    'adjari_3star_500.jpg',
    'adjari_5star_tubus.jpg',
    'adjari_7star_500.jpg',
    'dovbush_3star_500.jpg',
    'dovbush_4star_500.jpg',
    'dovbush_4star_round.jpg',
    'dovbush_honey_500.jpg',
]

for f in to_delete:
    path = os.path.join(refs, f)
    if os.path.exists(path):
        os.remove(path)
        print(f"  Deleted {f}")
    else:
        print(f"  Already gone: {f}")

print("\n=== Final file count ===")
remaining = sorted([f for f in os.listdir(refs) if f.endswith('.jpg')])
print(f"Total: {len(remaining)} reference images")
for f in remaining:
    print(f"  {f}")
```
